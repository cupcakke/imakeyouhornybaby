### The Executive Diagnosis: What Breaks the Native Illusion Right Now

Your codebase already has incredible ambition: WebGL2 refraction, spring physics math (`GlassKit.Spring`), an iOS layout bridge (`layoutBridge`, `inputBridge`, `hapticBridge`), branching dialogue trees, and offline IndexedDB caching. However, several critical architectural friction points and "uncanny valley" web artifacts immediately betray its web nature:

1. **The Three-Way Glass War**: You have **three conflicting glass systems** running simultaneously:
   - *System A*: CSS `backdrop-filter: blur(...)` and gradients.
   - *System B*: The fullscreen WebGL2 canvas (`MultiPassRenderer` in Script 9), which samples DOM rects only once every 600ms (`t - this._lastCollect >= 600`) capped at 48 shapes (`MAX_SHAPES = 48`). When a user scrolls, the WebGL refraction shapes lag behind the text by up to half a second, jittering and breaking.
   - *System C*: Script 4 mounts a separate React 18 `createRoot` inside *every single* `.lg, .bubble, .hcard` via `MutationObserver`, compiling SVG displacement filters with runtime Babel in the browser. Meanwhile, your CSS explicitly forces `.glass-host, .lg { background: transparent !important; backdrop-filter: none !important; }`, neutralizing these very filters.
2. **Missing Native Top Chrome (UINavigationBar)**: `#content` has no header bar at all. Chat bubbles scroll straight into the top notch / Dynamic Island area (`padding: calc(12px + var(--sat))`), text collides with white status bar icons, and there is no on-screen button to open history, view the model, or start a new chat. Users must know to swipe from an invisible 40px left-edge zone or hit `Cmd+K`.
3. **Transparent Input Collision**: `#ib-wrap` and `.ib-inner` have `background: transparent`. As messages scroll down, they pass directly *underneath* the input textarea and send button, causing message text and input text to bleed together illegibly.
4. **Broken User Message Expansion**: CSS defines `.user-message-text { max-height: 7.5em; overflow: hidden; }` and `.user-message-expand`, but `render()` in Script 7 *never creates* the expand button. Any user prompt longer than 7 lines is permanently truncated with zero way to read it.
5. **Streaming DOM Destruction**: Every 120ms (`scheduleStreamRender`), `body.innerHTML = markdown(...)` wipes out and rebuilds the active bubble, recalculating syntax highlighting for the entire message. This destroys active text selections, drops frames, and causes visual micro-stutters during streaming.
6. **Mobile Touch & Gesture Invisibility**:
   - History cards only listen to `contextmenu` (right-click). Because `-webkit-touch-callout: none` is set globally, mobile long-press does not trigger it.
   - User message edit buttons (`.msg-actions`) rely on `:hover`, rendering "Szerkesztés" completely inaccessible on touchscreens.
   - Swiping the sidebar does not dynamically ramp the backdrop opacity or check release velocity; a quick thumb flick under 80px snaps it shut instead of opening.
   - Lightbox has no 1-finger swipe-to-dismiss.
   - Scrolling down on the chat does not dismiss the software keyboard (`interactiveKeyboardDismiss`).

## 1. Unified Glass Architecture: Native CoreAnimation Compositing

### Eliminate the Runtime Babel & Multi-Root Overhead
- **The Problem**: Loading unpkg React 18, ReactDOM, and Babel Standalone (over 2.8 MB) to compile 95 KB of JSX at runtime on a mobile CPU drains battery, causes thermal throttling, and blocks the main thread at launch. Spawning hundreds of React roots via `MutationObserver` creates massive memory churn.
- **The Native Solution**: 
  - Standardize on hardware-accelerated **native CSS backdrop filters** with CoreAnimation layer promotion (`will-change: transform`, `transform: translateZ(0)`).
  - Apple’s WebKit engine renders `backdrop-filter: blur(28px) saturate(190%)` directly on the GPU compositor thread at 120 FPS ProMotion with near-zero CPU overhead.
  - Reserve the fullscreen WebGL2 canvas purely for **ambient, organic fluid background lighting** (`bgPass`), without attempting to mirror DOM element rects into an SDF scene that desyncs during kinetic scroll.

## 4. Native Gestures & Spring Motion Physics

### 1. Proportional Sidebar Drag with Velocity Flick
Update Script 8’s sidebar gestures with velocity tracking and 1:1 backdrop dimming:
```javascript
(function setupProportionalSidebarDrag() {
  let touchId = null;
  let startX = 0, lastX = 0, lastTime = 0, velocityX = 0;
  const sidebar = document.getElementById("sidebar");
  const backdrop = document.getElementById("sidebar-backdrop");
  const drawerWidth = () => Math.min(window.innerWidth * 0.82, 340);

  document.addEventListener("touchstart", (e) => {
    if (touchId !== null) return;
    const t = e.changedTouches[0];
    if (!sbOpen && t.clientX > 32) return; // Edge trigger zone
    if (e.target.closest("#ta, .cb, .lightbox-stage, input")) return;

    touchId = t.identifier;
    startX = t.clientX;
    lastX = t.clientX;
    lastTime = performance.now();
    velocityX = 0;
    sidebar.classList.add("dragging");
    backdrop.style.transition = "none";
  }, { passive: true });

  document.addEventListener("touchmove", (e) => {
    if (touchId === null) return;
    const t = Array.from(e.changedTouches).find(x => x.identifier === touchId);
    if (!t) return;

    const now = performance.now();
    const dt = Math.max(1, now - lastTime);
    velocityX = (t.clientX - lastX) / dt; // px/ms
    lastX = t.clientX;
    lastTime = now;

    const w = drawerWidth();
    let currentX = sbOpen ? Math.min(0, t.clientX - startX) : Math.min(0, -w + (t.clientX - startX));
    let progress = (currentX + w) / w; // 0.0 -> 1.0

    sidebar.style.transform = `translateX(${currentX}px)`;
    backdrop.style.opacity = Math.max(0, Math.min(1, progress)).toFixed(3);
    backdrop.style.pointerEvents = progress > 0.05 ? "auto" : "none";
  }, { passive: true });

  document.addEventListener("touchend", (e) => {
    if (touchId === null) return;
    const t = Array.from(e.changedTouches).find(x => x.identifier === touchId);
    if (!t) return;

    sidebar.classList.remove("dragging");
    sidebar.style.transform = "";
    backdrop.style.transition = "";
    backdrop.style.opacity = "";
    backdrop.style.pointerEvents = "";

    const w = drawerWidth();
    const finalDeltaX = t.clientX - startX;

    // Velocity-aware decision: a fast flick opens or closes regardless of distance
    if (!sbOpen) {
      if (velocityX > 0.35 || finalDeltaX > w * 0.4) openSB();
      else closeSB();
    } else {
      if (velocityX < -0.35 || finalDeltaX < -w * 0.3) closeSB();
      else openSB();
    }

    touchId = null;
  }, { passive: true });
})();
```

function triggerHaptic(style) {
  // 1. Native WKWebView Bridge
  if (window.webkit?.messageHandlers?.hapticBridge) {
    window.webkit.messageHandlers.hapticBridge.postMessage({ type: style, intensity: 1.0 });
    return;

  // 2. Web Vibration API Fallback (Android/PWA)
  if (navigator.vibrate) {
    switch (style) {

      case "impact-medium": navigator.vibrate(18); break;
      case "impact-heavy": navigator.vibrate(28); break;

      case "notification-success": navigator.vibrate([10, 30, 15]); break;
      case "notification-error": navigator.vibrate([20, 40, 20, 40, 30]); break;
    }
  }
}
